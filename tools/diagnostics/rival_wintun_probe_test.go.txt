//go:build windows && with_gvisor

// Independent test harness for ArmanLife's exported tunstack API. Copied only
// into an ephemeral checkout of the pinned upstream source by the CI job.
// This is an adapter/stack experiment, not either product's complete VPN.
package tunstack

import (
	"bytes"
	"crypto/sha256"
	"fmt"
	"io"
	"net"
	"net/netip"
	"os"
	"testing"
	"time"
)

func TestGozarComparisonLocalWintun(t *testing.T) {
	if os.Getenv("GITHUB_ACTIONS") != "true" || os.Getenv("GOZAR_RIVAL_PROBE") != "1" {
		t.Skip("disposable Windows CI only")
	}
	listener, err := net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	defer listener.Close()
	payload := bytes.Repeat([]byte("gozar-throughput-diagnostic-block"), 262144)
	sum := sha256.Sum256(payload)
	peer := make(chan error, 1)
	go func() {
		c, err := listener.Accept()
		if err != nil {
			peer <- err
			return
		}
		defer c.Close()
		_ = c.SetDeadline(time.Now().Add(70 * time.Second))
		h := sha256.New()
		if _, err = io.CopyN(h, c, int64(len(payload))); err != nil {
			peer <- err
			return
		}
		if !bytes.Equal(h.Sum(nil), sum[:]) {
			peer <- fmt.Errorf("upload checksum mismatch")
			return
		}
		if _, err = c.Write([]byte{0xa5}); err != nil {
			peer <- err
			return
		}
		b := make([]byte, 1)
		if _, err = io.ReadFull(c, b); err != nil || b[0] != 0x5a {
			peer <- fmt.Errorf("invalid download start: %v", err)
			return
		}
		_, err = io.Copy(c, bytes.NewReader(payload))
		peer <- err
	}()
	d, err := Open(Config{Name: "GozarComparison", Inet4: netip.MustParsePrefix("198.18.0.1/30"), MTU: 9000, Handler: Handler{
		DNS: func([]byte) ([]byte, error) { return nil, fmt.Errorf("DNS not used in this local diagnostic") },
		TCP: func(f *Flow) {
			if f.Destination != netip.MustParseAddrPort("198.18.0.80:443") {
				f.Refuse(fmt.Errorf("unexpected destination"))
				return
			}
			upstream, err := net.DialTimeout("tcp4", listener.Addr().String(), 5*time.Second)
			if err != nil {
				f.Refuse(err)
				return
			}
			local, err := f.Accept()
			if err != nil {
				upstream.Close()
				return
			}
			go func() {
				defer local.Close()
				defer upstream.Close()
				done := make(chan struct{})
				go func() { _, _ = io.CopyBuffer(upstream, local, make([]byte, 128*1024)); close(done) }()
				_, _ = io.CopyBuffer(local, upstream, make([]byte, 128*1024))
				upstream.Close()
				<-done
			}()
		},
	}})
	if err != nil {
		t.Fatal(err)
	}
	defer func() {
		if err := d.Close(); err != nil {
			t.Errorf("close: %v", err)
		}
	}()
	if err = d.Start(); err != nil {
		t.Fatal(err)
	}
	c, err := net.DialTimeout("tcp4", "198.18.0.80:443", 10*time.Second)
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close()
	_ = c.SetDeadline(time.Now().Add(60 * time.Second))
	started := time.Now()
	n, err := io.Copy(c, bytes.NewReader(payload))
	if err != nil || n != int64(len(payload)) {
		t.Fatalf("upload %d %v", n, err)
	}
	var ack [1]byte
	if _, err = io.ReadFull(c, ack[:]); err != nil || ack[0] != 0xa5 {
		t.Fatalf("upload acknowledgement %x %v", ack, err)
	}
	up := time.Since(started)
	started = time.Now()
	if _, err = c.Write([]byte{0x5a}); err != nil {
		t.Fatal(err)
	}
	h := sha256.New()
	n, err = io.CopyN(h, c, int64(len(payload)))
	down := time.Since(started)
	if err != nil || n != int64(len(payload)) || !bytes.Equal(h.Sum(nil), sum[:]) {
		t.Fatalf("download %d %v digest=%t", n, err, bytes.Equal(h.Sum(nil), sum[:]))
	}
	if err = <-peer; err != nil {
		t.Fatal(err)
	}
	rate := func(d time.Duration) float64 { return float64(len(payload)*8) / d.Seconds() / 1e6 }
	t.Logf("RIVAL LOCAL WINTUN: bytes=%d upload=%.2f Mbit/s (%s) download=%.2f Mbit/s (%s); TCP stack plus local peer only; no VPN relay/WAN/Telegram", len(payload), rate(up), up, rate(down), down)
}
